/* Math Answer Normalization and Evaluation */ function normalizeMath(str) { if (!str) return ""; return str.toString() // LaTeX 문법 및 괄호 제거 .replace(/\\\(/g, '').replace(/\\\)/g, '') .replace(/\\left/g, '').replace(/\\right/g, '') .replace(/\\frac\{([^}]+)\}\{([^}]+)\}/g, '$1/$2') .replace(/[\$\s]/g, '') // 변수 표기 제거 (ex: x=, y=, z=) .replace(/[a-zA-Z]\s*=\s*/g, '') // 접속어 표준화 (또는, or, and 등을 쉼표로 통일) .replace(/(또는|or|and|\|\|)/g, ',') .toLowerCase(); } function extractNumbersOrValues(str) { const normalized = normalizeMath(str); // 음수, 소수, 분수 형태의 숫자를 모두 추출 const matches = normalized.match(/-?\d+(\.\d+)?(\/-?\d+(\.\d+)?)?/g); if (!matches) return null; return matches.map(m => { if (m.includes('/')) { const parts = m.split('/'); return parseFloat(parts[0]) / parseFloat(parts[1]); } return parseFloat(m); }); } function checkAnswer() { const userInput = document.getElementById('user-answer-input').value.trim(); const feedbackEl = document.getElementById('answer-feedback'); const correctAnswer = currentProblemData?.correctAnswer || ""; if (!userInput) { feedbackEl.className = "text-sm font-bold text-amber-600"; feedbackEl.innerText = "답을 입력해 주세요."; return; } const userNorm = normalizeMath(userInput); const ansNorm = normalizeMath(correctAnswer); // 1차: 단순 문자열 정규화 비교 let isCorrect = (userNorm === ansNorm); // 2차: 추출된 숫자 집합 비교 (순서 무관) if (!isCorrect) { const userVals = extractNumbersOrValues(userInput); const ansVals = extractNumbersOrValues(correctAnswer); if (userVals && ansVals && userVals.length === ansVals.length) { // 오름차순 정렬 후 각각 오차범위 내 비교 const sortedUser = [...userVals].sort((a, b) => a - b); const sortedAns = [...ansVals].sort((a, b) => a - b); isCorrect = sortedUser.every((v, idx) => Math.abs(v - sortedAns[idx]) < 0.001); } } if (isCorrect) { feedbackEl.className = "text-sm font-bold text-emerald-600"; feedbackEl.innerHTML = ' 정답입니다! 아주 잘하셨어요.'; } else { feedbackEl.className = "text-sm font-bold text-rose-500"; feedbackEl.innerHTML = ' 다시 한번 계산해 보세요.'; } }
